Target "Linux"
UseLib "library/stdlib.wlib"

@data {
    WSTR str_hi, 'Hello, World!', 13, 10
    WSTR_LEN str_hi
}

let str_hru: str = "How are you?"
let str_new: str = "What are you doing?"

fn syscall(a: any, b: any, c: any) {
    asm {
        mov rax, {a}
        mov rdi, {b}
        lea rsi, [{c}]
        mov rdx, {c}_len
        syscall
    }
}

fn main() {
    syscall(1, 1, str_hi)
    syscall(1, 1, str_hru)
    exit()
}
